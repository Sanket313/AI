def alphabeta(d, i, maxplayer, values, alpha, beta):
    if d == 3:
        return values[i]

    if maxplayer:
        best = -999
        for j in range(2):
            v = alphabeta(d+1, i*2+j, False, values, alpha, beta)
            best = max(best, v)
            alpha = max(alpha, best)
            if beta <= alpha:
                break
        return best
    else:
        best = 999
        for j in range(2):
            v = alphabeta(d+1, i*2+j, True, values, alpha, beta)
            best = min(best, v)
            beta = min(beta, best)
            if beta <= alpha:
                break
        return best

values = [3, 5, 6, 9, 1, 2, 0, -1]
print("Best Value:", alphabeta(0, 0, True, values, -999, 999))
