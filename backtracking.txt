board = []

def solve(row):
    if row == 8:
        return True

    for col in range(8):
        safe = True
        for r in range(row):
            if board[r] == col or abs(board[r] - col) == row - r:
                safe = False

        if safe:
            board.append(col)
            if solve(row + 1):
                return True
            board.pop()

    return False

solve(0)
print("Solution:", board)


Practical 6: 8-Queens Using Backtracking
Aim: To solve the 8-Queens problem using Backtracking.
Logic: Place one queen in each row. If a position is unsafe or a later row has no safe position, remove the previous queen and try another column.

Algorithm
1.	Start with an empty board.
2.	Try a column in the current row.
3.	Check whether the column and diagonals are safe.
4.	Place a queen and solve the next row.
5.	If the next row cannot be solved, remove the queen and try another column.
6.	When all 8 rows are filled, print the solution.

    IF row == 8

            IF SOLVE(row + 1)

            REMOVE queen
    RETURN FALSE
