board = []

def solve(row):
    if row == 8:
        return True

    for col in range(8):
        safe = True
        for r in range(row):
            if board[r] == col or abs(board[r] - col) == row - r:
                safe = False

        if safe:
            board.append(col)
            if solve(row + 1):
                return True
            board.pop()

    return False

solve(0)
print("Solution:", board)
